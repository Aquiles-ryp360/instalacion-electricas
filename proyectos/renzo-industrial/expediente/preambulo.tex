\newcommand{\PortadaCapitulo}[1]{%
  \vspace*{\fill}%
  \begin{center}
    {\LARGE\bfseries\color{navyblue} CAPÍTULO \Roman{chapter}\par}
    \vspace{1.2cm}
    {\color{navyblue}\rule{0.78\textwidth}{1.5pt}}\par
    \vspace{0.9cm}
    \begin{minipage}{0.92\textwidth}
      \Centering
      \pretolerance=10000
      \tolerance=10000
      \hyphenpenalty=10000
      \exhyphenpenalty=10000
      \emergencystretch=2em
      {\LARGE\bfseries\color{navyblue} #1\par}
    \end{minipage}\par
    \vspace{0.9cm}
    {\color{navyblue}\rule{0.78\textwidth}{1.5pt}}
  \end{center}%
  \vspace*{\fill}%
  \clearpage
}
