\chapter{Memoria Descriptiva}

\section{Introducción y Justificación del Proyecto}
El presente proyecto comprende el diseño, dimensionamiento electromecánico y presupuesto integral de la \textbf{Línea Primaria en Media Tensión (22.9 kV, 3Ø)} y la \textbf{Subestación de Distribución Biposte Proyectada (SED-01, 100 kVA)}, requeridas para suministrar energía eléctrica continua, confiable y segura al sector rural \textbf{Chifrón}, situado en la península de Capachica, provincia y departamento de Puno.

Este expediente técnico se desarrolla bajo la cátedra de la asignatura de especialidad \textbf{MEC330 --- Instalaciones Eléctricas II} de la Escuela Profesional de Ingeniería Mecánica Eléctrica de la Universidad Nacional del Altiplano (UNA Puno), bajo la tutela de los docentes ingenieros \textbf{Leonidas Vilca Callata} y \textbf{Marcos José Villanueva Cornejo}.

\section{Objetivos del Proyecto}
\begin{itemize}
    \item \textbf{Objetivo General:} Dotar de infraestructura eléctrica integral de media tensión ($22.9\text{ kV}$) y redes secundarias en baja tensión ($380/220\text{ V}$) al sector Chifrón, beneficiando directamente a \textbf{43 viviendas censadas} proyectadas a \textbf{78 abonados al año 20}, fomentando el desarrollo socioeconómico y el potencial ecoturístico lacustre.
    \item \textbf{Objetivos Específicos:}
    \begin{itemize}
        \item Ubicar y dimensionar la \textbf{Subestación de Distribución Biposte (SED SAB-1)} en la estructura proyectada $\mathbf{E-04}$ ($E = 412,759.20\text{ m}, N = 8,270,976.22\text{ m}$, Cota $3,839.56\text{ msnm}$, Progresiva $0+305.21\text{ m}$), alimentada desde el \textbf{Punto de Derivación MT E-01} ($E = 412,606.01\text{ m}, N = 8,270,712.24\text{ m}$, Cota $3,924.17\text{ msnm}$, Lat: $-15.640067^\circ\text{ S}$, Lon: $-69.815400^\circ\text{ W}$, WGS-84 Zona 19S), con una potencia comercial normalizada de $\mathbf{100\text{ kVA}}$ ($22.9 / 0.38 - 0.22\text{ kV}$, Dyn5).
        \item Estructurar la red secundaria de baja tensión en \textbf{exactamente dos (02) circuitos de distribución}:
        \begin{itemize}
            \item \textbf{Circuito C-1 (Troncal Norte + C31):} Longitud de $820.0\text{ m}$ con conductor autoportante CAAI $3\times35+1\times25+1\times16\text{ mm}^2$, abasteciendo a 31 viviendas censadas ($56$ abonados proyectados) hacia la franja de playa y muelle, integrando la vivienda $C31$ por derivación directa tras la eliminación técnica de la rama central redundante.
            \item \textbf{Circuito C-2 (Troncal Carretera al Fondo):} Longitud de $456.0\text{ m}$ con conductor CAAI $3\times25+1\times25+1\times16\text{ mm}^2$, abasteciendo a 12 viviendas censadas ($22$ abonados proyectados) que se extienden de manera continua a lo largo de la vía carrozable principal hasta el extremo sur y fondo de la carretera ($C32$ a $C43$).
        \end{itemize}
        \item Integrar planos normalizados con calidad AutoCAD pura en formato A3 y adoptar las láminas oficiales de detalle electromecánico aprobadas por la concesionaria \textbf{Electro Puno S.A.A.}
    \end{itemize}
\end{itemize}

\chapter{Cálculos Justificativos Eléctricos y Mecánicos}

\section{Metodología Oficial de Proyección y Cuadro de Cargas}
El dimensionamiento de la demanda eléctrica sigue la formulación oficial del curso \textit{MEC330 --- Instalaciones Eléctricas II} (FIME $\cdot$ UNA Puno).

\subsection{Parámetros de Partida}
\begin{itemize}
    \item \textbf{Usuarios Iniciales ($NU_0$):} $43\text{ viviendas}$ censadas en campo ($C1$ a $C43$).
    \item \textbf{Tasa de Crecimiento Anual ($TC$):} $3.0\%\ (0.03)$.
    \item \textbf{Horizonte de Proyección ($n$):} $20\text{ años}$ ($2026 - 2046$).
    \item \textbf{Calificación Eléctrica Asignada ($CE$):} $0.90\text{ kW / vivienda}$.
    \item \textbf{Tensión BT y Factor de Potencia:} $V = 380/220\text{ V}$, $\cos\phi = 0.90$.
    \item \textbf{Potencia por Luminaria de Alumbrado Público:} $P_{\text{lum}} = 60\text{ W} = 0.06\text{ kW}$.
\end{itemize}

\subsection{Formulación Matemática Oficial}
\begin{enumerate}
    \item \textbf{Proyección de Usuarios (Servicio Particular):}
    \begin{equation}
        NU_f = NU_0 \cdot (1 + TC)^n = 43 \cdot (1 + 0.03)^{20} = 43 \cdot 1.80611 = 77.66 \approx \mathbf{78\text{ usuarios}}
    \end{equation}
    \item \textbf{Demanda del Servicio Particular:}
    \begin{equation}
        Pot_{\text{part}} = NU_f \cdot CE = 78 \cdot 0.90\text{ kW} = \mathbf{70.20\text{ kW}}
    \end{equation}
    \item \textbf{Número de Luminarias de Alumbrado Público:}
    \begin{equation}
        N^\circ lum = \frac{3.3 \cdot NU_f}{18} = \frac{3.3 \cdot 78}{18} = 14.30 \approx \mathbf{15\text{ luminarias}}
    \end{equation}
    \item \textbf{Potencia de Alumbrado Público:}
    \begin{equation}
        Pot_{\text{alum}} = N^\circ lum \cdot 0.06\text{ kW} = 15 \cdot 0.06 = \mathbf{0.90\text{ kW}}
    \end{equation}
    \item \textbf{Demanda Máxima Total Activa:}
    \begin{equation}
        Total_{\text{kW}} = Pot_{\text{part}} + Pot_{\text{alum}} = 70.20 + 0.90 = \mathbf{71.10\text{ kW}}
    \end{equation}
    \item \textbf{Corriente Nominal Total en Baja Tensión ($3\phi$, $380\text{ V}$):}
    \begin{equation}
        I = \frac{Total_{\text{kW}} \cdot 1000}{\sqrt{3} \cdot 380 \cdot \cos\phi} = \frac{71.10 \cdot 1000}{1.732 \cdot 380 \cdot 0.90} = \mathbf{120.03\text{ A}}
    \end{equation}
    \item \textbf{Potencia Aparente Requerida y Selección del Transformador:}
    \begin{equation}
        Pot_{\text{kVA}} = \frac{Total_{\text{kW}}}{\cos\phi} = \frac{71.10}{0.90} = \mathbf{79.00\text{ kVA}}
    \end{equation}
\end{enumerate}

\subsection{Selección Definitiva del Transformador Comercial}
Para cubrir una demanda aparente de $\mathbf{79.00\text{ kVA}}$, se selecciona en la serie normalizada de transformadores de distribución trifásicos en el Perú ($25, 37.5, 50, 75, 100, 160\text{ kVA}$) la potencia comercial inmediata superior:
\begin{equation}
    \mathbf{S_{\text{nom}} = 100\text{ kVA}} \qquad (22.9 / 0.38 - 0.22\text{ kV}, \text{ conexión Dyn5})
\end{equation}
El factor de utilización al año 20 es de $79.00 / 100 = 79.0\%$, otorgando una reserva de capacidad técnica del $\mathbf{21.0\%}$ para absorber corrientes de arranque de motores y crecimiento no previsto.

\section{Cálculo Eléctrico de la Línea Primaria (22.9 kV)}
\subsection{Parámetros Eléctricos del Conductor AAAC 35 mm\texorpdfstring{$^2$}{2}}
\begin{itemize}
    \item Resistencia a $20^\circ\text{C}$: $R_{20} = 0.963\ \Omega/\text{km}$.
    \item Resistencia corregida a $50^\circ\text{C}$: $R_{50} = R_{20} \cdot [1 + 0.0036 \cdot (50 - 20)] = 0.986\ \Omega/\text{km}$.
    \item Reactancia inductiva (para $DMG = 1.20\text{ m}$ y $r = 3.75\text{ mm}$):
    \begin{equation}
        X = 2\pi f \cdot \left[ 0.05 + 0.2 \ln\left(\frac{DMG}{r}\right) \right] \cdot 10^{-3} = 0.385\ \Omega/\text{km}
    \end{equation}
    \item Longitud de línea: $L = 0.817\text{ km}$.
\end{itemize}

\subsection{Caída de Tensión Porcentual (\texorpdfstring{$\Delta V\%$}{Delta V\%})}
Para la potencia nominal de la SED ($S = 100\text{ kVA}$ a $22.9\text{ kV}$):
\begin{equation}
    I_{\text{MT}} = \frac{100\text{ kVA}}{\sqrt{3} \cdot 22.9\text{ kV}} = 2.52\text{ A}
\end{equation}
\begin{equation}
    \Delta V = \sqrt{3} \cdot I_{\text{MT}} \cdot L \cdot (R \cos\phi + X \sin\phi) = \sqrt{3} \cdot 2.52 \cdot 0.817 \cdot (0.986 \cdot 0.9 + 0.385 \cdot 0.436) = 3.76\text{ V}
\end{equation}
\begin{equation}
    \Delta V\% = \frac{\Delta V}{V_{\text{nom}}} \cdot 100 = \frac{3.76}{22,900} \cdot 100 = \mathbf{0.016\%} \le 5.0\% \quad \textbf{(AMPLIAMENTE SATISFACTORIO)}
\end{equation}

\end{equation}

    \begin{equation}
        NU_f = 43 \cdot (1 + 0.03)^{20} = 43 \cdot 1.80611 = 77.66 \implies 78\text{ abonados}
    \end{equation}
    \item \textbf{Potencia particular:} $78 \times 0.90\text{ kW} = 70.20\text{ kW}$.
    \item \textbf{Alumbrado público:} $N^\circ_{\text{lum}} = \frac{3.3 \cdot 78}{18} = 14.3 \implies 15\text{ luminarias LED 60W} \implies 0.90\text{ kW}$.

\section{Auditoría de Puestas a Tierra y Seguridad Humana}
\subsection{Verificación Teórica por Fórmula de Dwight}
La resistividad promedio del suelo en la falda de cerro y ribera de Chifrón se estima en $\rho \approx 200\ \Omega\cdot\text{m}$. La resistencia teórica de un electrodo vertical único de cobre de $5/8'' \times 2.40\text{ m}$ sin tratamiento químico resulta:
\begin{equation}
    R = \frac{\rho}{2\pi L} \left[ \ln\left( \frac{4L}{d} \right) - 1 \right] = \frac{200}{2\pi \cdot 2.40} \left[ \ln\left( \frac{4 \cdot 2.40}{0.01588} \right) - 1 \right] = 71.74\ \Omega
\end{equation}

\subsection{Observación Crítica y Solución de Ingeniería}
\textbf{Observación Crítica:} El CNE Suministro (Regla 362) y la Norma DGE establecen:
\begin{itemize}
    \item Resistencia en Subestaciones de Distribución (SED MT): $\mathbf{R \le 10.0\ \Omega}$.
    \item Resistencia en Redes Secundarias y Neutro (BT): $\mathbf{R \le 25.0\ \Omega}$.
\end{itemize}
Un valor natural de $71.74\ \Omega$ pondría en riesgo la seguridad de las personas y equipos.\\
\textbf{Mejora Implementada:}
\begin{enumerate}
    \item \textbf{Para Puestas a Tierra de BT (PAT-1):} Se prescribe la excavación de un pozo de $0.80\text{ m}\ \phi \times 2.60\text{ m}$ de profundidad, sustitución por tierra de cultivo cernida y aplicación de \textbf{30 kg de bentonita sódica de alta conductividad} más dosis gelificante Thor-Gel, reduciendo la resistividad aparente del terreno a $\rho_{\text{ap}} \le 35\ \Omega\cdot\text{m}$. La resistencia resultante se reduce a:
    \begin{equation}
        R_{\text{PAT-1}} \le 12.55\ \Omega \le \mathbf{25.0\ \Omega} \quad \textbf{(AMPLIAMENTE CUMPLE)}
    \end{equation}
    \item \textbf{Para la SED Biposte (PAT-3):} Se diseña una malla de \textbf{dos (02) electrodos verticales de cobre de $5/8'' \times 2.40\text{ m}$ en paralelo}, separados a una distancia de $S = 4.80\text{ m}$ (el doble de su longitud para evitar interferencia mutua de conos de potencial), unidos mediante conductor de cobre desnudo de $35\text{ mm}^2$ embebido en zanja con bentonita sódica. La resistencia equivalente del conjunto es:
    \begin{equation}
        R_{\text{SED}} \le 6.80\ \Omega \le \mathbf{10.0\ \Omega} \quad \textbf{(CUMPLE ESTRICTAMENTE)}
    \end{equation}
\end{enumerate}
